import { Injectable, Logger } from '@nestjs/common'
import { ConfigService } from '@nestjs/config'
import axios from 'axios'

@Injectable()
export class ZapiService {
  private logger = new Logger('ZapiService')
  private baseUrl: string

  constructor(private config: ConfigService) {
    this.baseUrl = config.get('ZAPI_BASE_URL')
  }

  // Enviar mensagem de texto
  async enviarTexto(
    instanceId: string,
    instanceToken: string,
    telefone: string,
    mensagem: string,
  ): Promise<boolean> {
    try {
      const url = `${this.baseUrl}/instances/${instanceId}/token/${instanceToken}/send-text`
      const phone = this.formatarTelefone(telefone)

      await axios.post(url, {
        phone,
        message: mensagem,
      })

      this.logger.log(`Mensagem enviada para ${phone} via instância ${instanceId}`)
      return true
    } catch (e) {
      this.logger.error(`Erro ao enviar mensagem: ${e.message}`, e.stack)
      return false
    }
  }

  // Enviar mensagem com botões (para confirmação de presença)
  async enviarBotoes(
    instanceId: string,
    instanceToken: string,
    telefone: string,
    mensagem: string,
    botoes: { id: string; label: string }[],
  ): Promise<boolean> {
    try {
      const url = `${this.baseUrl}/instances/${instanceId}/token/${instanceToken}/send-button-list`
      const phone = this.formatarTelefone(telefone)

      await axios.post(url, {
        phone,
        message: mensagem,
        buttonList: {
          buttons: botoes.map(b => ({
            id: b.id,
            label: b.label,
          })),
        },
      })

      this.logger.log(`Botões enviados para ${phone}`)
      return true
    } catch (e) {
      // Fallback: se botões não funcionarem, enviar texto simples
      this.logger.warn(`Botões falharam, enviando texto: ${e.message}`)
      const textoFallback = `${mensagem}\n\nResponda:\n1 - Confirmar presença\n2 - Cancelar`
      return this.enviarTexto(instanceId, instanceToken, telefone, textoFallback)
    }
  }

  // Gerar QR Code para conectar instância
  async gerarQrCode(instanceId: string, instanceToken: string): Promise<string | null> {
    try {
      const url = `${this.baseUrl}/instances/${instanceId}/token/${instanceToken}/qr-code/image`
      const { data } = await axios.get(url)
      return data?.value ?? null // retorna base64 da imagem
    } catch (e) {
      this.logger.error(`Erro ao gerar QR: ${e.message}`)
      return null
    }
  }

  // Verificar status da conexão
  async verificarStatus(instanceId: string, instanceToken: string): Promise<string> {
    try {
      const url = `${this.baseUrl}/instances/${instanceId}/token/${instanceToken}/status`
      const { data } = await axios.get(url)
      return data?.connected ? 'conectado' : 'desconectado'
    } catch (e) {
      this.logger.error(`Erro ao verificar status: ${e.message}`)
      return 'erro'
    }
  }

  // Desconectar instância
  async desconectar(instanceId: string, instanceToken: string): Promise<void> {
    try {
      const url = `${this.baseUrl}/instances/${instanceId}/token/${instanceToken}/disconnect`
      await axios.post(url)
    } catch (e) {
      this.logger.error(`Erro ao desconectar: ${e.message}`)
    }
  }

  // Criar nova instância Z-API para um profissional
  // NOTA: depende do plano Z-API. Verificar se API permite criar instância
  // programaticamente ou se precisa ser manual no painel.
  // Se manual: profissional informa instanceId e token nas configurações.

  // Formatar telefone brasileiro para formato Z-API
  private formatarTelefone(tel: string): string {
    const digits = tel.replace(/\D/g, '')
    // Se já tem 55 (DDI): retornar como está
    if (digits.startsWith('55') && digits.length >= 12) return digits
    // Se tem 10-11 dígitos (DDD + número): adicionar 55
    if (digits.length >= 10 && digits.length <= 11) return `55${digits}`
    return digits
  }
}

import { Controller, Post, Body, Headers, Logger, UnauthorizedException } from '@nestjs/common'
import { ConfigService } from '@nestjs/config'
import { ApiTags, ApiOperation, ApiExcludeEndpoint } from '@nestjs/swagger'
import { WebhooksService } from './webhooks.service'

@ApiTags('Webhooks')
@Controller('api/webhooks')
export class WebhooksController {
  private logger = new Logger('Webhooks')

  constructor(
    private service: WebhooksService,
    private config: ConfigService,
  ) {}

  @Post('zapi')
  @ApiExcludeEndpoint() // não expor no Swagger público
  async receberZapi(
    @Body() body: any,
    @Headers('x-webhook-secret') secret: string,
  ) {
    // Validar secret (opcional — depende da config Z-API)
    const expectedSecret = this.config.get('ZAPI_WEBHOOK_SECRET')
    if (expectedSecret && secret !== expectedSecret) {
      this.logger.warn('Webhook com secret inválido')
      throw new UnauthorizedException()
    }

    this.logger.log(`Webhook Z-API: ${body?.type ?? 'unknown'}`)
    await this.service.processar(body)
    return { ok: true }
  }
}

import { Injectable, Logger } from '@nestjs/common'
import { RespostasService } from '../respostas/respostas.service'

@Injectable()
export class WebhooksService {
  private logger = new Logger('WebhooksService')

  constructor(private respostas: RespostasService) {}

  async processar(payload: any): Promise<void> {
    // Z-API envia diferentes tipos de evento
    // Nos interessa: mensagens recebidas (respostas do cliente)
    const tipo = payload?.type

    if (tipo === 'ReceivedCallback' || tipo === 'MessageStatusCallback') {
      // Mensagem recebida de um cliente
      const phone = payload?.phone ?? payload?.from
      const text = payload?.text?.message ?? payload?.body ?? ''
      const instanceId = payload?.instanceId

      if (phone && text) {
        await this.respostas.processarResposta(instanceId, phone, text.trim())
      }
    }

    // Status de conexão da instância
    if (tipo === 'StatusInstanceCallback') {
      const instanceId = payload?.instanceId
      const connected = payload?.connected
      if (instanceId) {
        await this.respostas.atualizarStatusConexao(instanceId, connected)
      }
    }
  }
}

import { Injectable, Logger } from '@nestjs/common'
import { ConfigService } from '@nestjs/config'
import { createClient, SupabaseClient } from '@supabase/supabase-js'
import { ZapiService } from '../zapi/zapi.service'

@Injectable()
export class NotificacoesService {
  private logger = new Logger('Notificacoes')
  private supabase: SupabaseClient

  constructor(
    private zapi: ZapiService,
    private config: ConfigService,
  ) {
    this.supabase = createClient(
      config.get('SUPABASE_URL'),
      config.get('SUPABASE_SERVICE_ROLE_KEY'),
    )
  }

  // Buscar dados de conexão WhatsApp do profissional
  private async getWppConfig(profissionalId: string) {
    const { data } = await this.supabase
      .from('profissionais')
      .select('wpp_instance_id, wpp_token, wpp_status, nome, whatsapp')
      .eq('id', profissionalId)
      .single()

    if (!data?.wpp_instance_id || !data?.wpp_token || data?.wpp_status !== 'conectado') {
      this.logger.warn(`Profissional ${profissionalId} sem WhatsApp conectado`)
      return null
    }
    return data
  }

  }

  // Formatação de data/hora para exibição
  private formatarDataHora(iso: string): string {
    const d = new Date(iso)
    const dia = d.toLocaleDateString('pt-BR', {
      weekday: 'long', day: '2-digit', month: '2-digit', year: 'numeric',
      timeZone: 'America/Sao_Paulo',
    })
    const hora = d.toLocaleTimeString('pt-BR', {
      hour: '2-digit', minute: '2-digit',
      timeZone: 'America/Sao_Paulo',
    })
    return `${dia} às ${hora}`
  }

import { Injectable, Logger } from '@nestjs/common'
import { Cron, CronExpression } from '@nestjs/schedule'
import { ConfigService } from '@nestjs/config'
import { createClient, SupabaseClient } from '@supabase/supabase-js'
import { NotificacoesService } from '../notificacoes/notificacoes.service'

@Injectable()
export class LembretesService {
  private logger = new Logger('Lembretes')
  private supabase: SupabaseClient

  constructor(
    private notificacoes: NotificacoesService,
    private config: ConfigService,
  ) {
    this.supabase = createClient(
      config.get('SUPABASE_URL'),
      config.get('SUPABASE_SERVICE_ROLE_KEY'),
    )
  }

  // Roda a cada 15 minutos
  @Cron('0 */15 * * * *')
  async verificarLembretes(): Promise<void> {
    this.logger.log('Verificando lembretes...')

    // Buscar profissionais com lembrete configurado e WhatsApp conectado
    const { data: profissionais } = await this.supabase
      .from('profissionais')
      .select('id, lembrete_horas, cancelamento_auto_cliente, wpp_status')
      .not('lembrete_horas', 'is', null)
      .eq('wpp_status', 'conectado')

    if (!profissionais?.length) {
      this.logger.log('Nenhum profissional com lembrete configurado')
      return
    }

    for (const prof of profissionais) {
      await this.processarLembretesProfissional(prof)
    }
  }

  private async processarLembretesProfissional(prof: {
    id: string
    lembrete_horas: number
    cancelamento_auto_cliente: boolean
  }): Promise<void> {
    // Calcular janela: agendamentos que começam entre agora+lembreteHoras-15min e agora+lembreteHoras+15min
    const agora = new Date()
    const janelaInicio = new Date(agora.getTime() + (prof.lembrete_horas * 60 - 15) * 60000)
    const janelaFim = new Date(agora.getTime() + (prof.lembrete_horas * 60 + 15) * 60000)

    const { data: agendamentos } = await this.supabase
      .from('agendamentos')
      .select('id, cliente_nome, cliente_wpp, servico_id, data_hora, profissional_id')
      .eq('profissional_id', prof.id)
      .eq('status', 'confirmado')
      .eq('lembrete_enviado', false)
      .gte('data_hora', janelaInicio.toISOString())
      .lte('data_hora', janelaFim.toISOString())

    if (!agendamentos?.length) return

    this.logger.log(`${agendamentos.length} lembrete(s) para profissional ${prof.id}`)

    for (const ag of agendamentos) {
      // Buscar nome do serviço
      const { data: servico } = await this.supabase
        .from('servicos')
        .select('nome')
        .eq('id', ag.servico_id)
        .single()

      const enviado = await this.notificacoes.enviarLembrete({
        id: ag.id,
        profissional_id: ag.profissional_id,
        cliente_wpp: ag.cliente_wpp,
        cliente_nome: ag.cliente_nome,
        servico_nome: servico?.nome ?? 'Serviço',
        data_hora: ag.data_hora,
        cancelamento_auto: prof.cancelamento_auto_cliente,
      })

      if (enviado) {
        // Marcar lembrete como enviado
        await this.supabase
          .from('agendamentos')
          .update({ lembrete_enviado: true })
          .eq('id', ag.id)
      }
    }
  }
}

import { Injectable, Logger } from '@nestjs/common'
import { ConfigService } from '@nestjs/config'
import { createClient, SupabaseClient } from '@supabase/supabase-js'
import { NotificacoesService } from '../notificacoes/notificacoes.service'

@Injectable()
export class RespostasService {
  private logger = new Logger('Respostas')
  private supabase: SupabaseClient

  constructor(
    private notificacoes: NotificacoesService,
    private config: ConfigService,
  ) {
    this.supabase = createClient(
      config.get('SUPABASE_URL'),
      config.get('SUPABASE_SERVICE_ROLE_KEY'),
    )
  }

  async processarResposta(instanceId: string, phone: string, texto: string): Promise<void> {
    // Normalizar texto: pegar primeiro caractere ou ID do botão
    const resposta = texto.toLowerCase().trim()
    const ehConfirmacao = resposta === '1' || resposta.includes('confirmar')
    const ehCancelamento = resposta === '2' || resposta.includes('cancelar')

    if (!ehConfirmacao && !ehCancelamento) {
      this.logger.log(`Resposta não reconhecida de ${phone}: "${texto}"`)
      return
    }

    // Buscar profissional pela instância Z-API
    const { data: prof } = await this.supabase
      .from('profissionais')
      .select('id, cancelamento_auto_cliente')
      .eq('wpp_instance_id', instanceId)
      .single()

    if (!prof) {
      this.logger.warn(`Profissional não encontrado para instância ${instanceId}`)
      return
    }

    // Buscar agendamento mais recente do cliente com lembrete enviado
    const phoneDigits = phone.replace(/\D/g, '').replace(/^55/, '')
    const { data: agendamentos } = await this.supabase
      .from('agendamentos')
      .select('id, profissional_id, cliente_nome, servico_id, data_hora, cliente_wpp, status')
      .eq('profissional_id', prof.id)
      .eq('status', 'confirmado')
      .eq('lembrete_enviado', true)
      .is('confirmacao_presenca', null)
      .order('data_hora', { ascending: true })
      .limit(10)

    // Encontrar agendamento cujo cliente_wpp bate com o phone
    const ag = agendamentos?.find(a => {
      const agPhone = a.cliente_wpp.replace(/\D/g, '')
      return phoneDigits.endsWith(agPhone) || agPhone.endsWith(phoneDigits.slice(-8))
    })

    if (!ag) {
      this.logger.warn(`Nenhum agendamento encontrado para ${phone}`)
      return
    }

    if (ehConfirmacao) {
      // Cliente confirmou presença
      await this.supabase
        .from('agendamentos')
        .update({ confirmacao_presenca: 'confirmou' })
        .eq('id', ag.id)

      this.logger.log(`Presença confirmada: agendamento ${ag.id}`)
    }

    if (ehCancelamento) {
      if (prof.cancelamento_auto_cliente) {
        // Cancelar automaticamente
        await this.supabase
          .from('agendamentos')
          .update({ status: 'cancelado', confirmacao_presenca: 'cancelou' })
          .eq('id', ag.id)

        this.logger.log(`Cancelamento automático: agendamento ${ag.id}`)

        // Notificar profissional
        const { data: servico } = await this.supabase
          .from('servicos')
          .select('nome')
          .eq('id', ag.servico_id)
          .single()

        await this.notificacoes.notificarCancelamentoCliente({
          profissional_id: ag.profissional_id,
          cliente_nome: ag.cliente_nome,
          servico_nome: servico?.nome ?? 'Serviço',
          data_hora: ag.data_hora,
        })
      } else {
        // Cancelamento automático desabilitado: ignorar resposta "2"
        this.logger.log(`Cancelamento ignorado (desabilitado): agendamento ${ag.id}`)
      }
    }
  }

  // Atualizar status de conexão WhatsApp do profissional
  async atualizarStatusConexao(instanceId: string, connected: boolean): Promise<void> {
    const status = connected ? 'conectado' : 'desconectado'
    await this.supabase
      .from('profissionais')
      .update({ wpp_status: status })
      .eq('wpp_instance_id', instanceId)

    this.logger.log(`Instância ${instanceId}: ${status}`)
  }
}

@ApiTags('WhatsApp')
@Controller('api/whatsapp')
export class ZapiController {
  constructor(
    private zapiService: ZapiService,
    private config: ConfigService,
  ) {}

  @Post('qr-code')
  @ApiBearerAuth()
  @UseGuards(SupabaseAuthGuard)
  @ApiOperation({ summary: 'Gerar QR Code para conectar WhatsApp' })
  async qrCode(@CurrentUser() user: any) {
    // Buscar instância do profissional
    const supabase = createClient(
      this.config.get('SUPABASE_URL'),
      this.config.get('SUPABASE_SERVICE_ROLE_KEY'),
    )
    const { data: prof } = await supabase
      .from('profissionais')
      .select('wpp_instance_id, wpp_token')
      .eq('user_id', user.id)
      .single()

    if (!prof?.wpp_instance_id || !prof?.wpp_token) {
      throw new BadRequestException(
        'Configure o Instance ID e Token da Z-API nas configurações.'
      )
    }

    const qrCode = await this.zapiService.gerarQrCode(prof.wpp_instance_id, prof.wpp_token)
    if (!qrCode) throw new InternalServerErrorException('Erro ao gerar QR Code')
    return { qrCode }
  }
