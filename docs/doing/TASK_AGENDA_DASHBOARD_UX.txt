  semanaOffset = signal(0)  // 0 = semana atual, -1 = anterior, 1 = próxima

  semana = computed(() => this.gerarSemana(this.semanaOffset()))

  semanaAnterior() { this.semanaOffset.update(v => v - 1) }
  proximaSemana()  { this.semanaOffset.update(v => v + 1) }

  private gerarSemana(offset = 0) {
    const hoje = new Date()
    const diasLetras = ['D','S','T','Q','Q','S','S']
    const baseDate = new Date(hoje)
    baseDate.setDate(hoje.getDate() - hoje.getDay() + offset * 7)
    return Array.from({ length: 7 }, (_, i) => {
      const d = new Date(baseDate)
      d.setDate(baseDate.getDate() + i)
      const k = d.toDateString()
      const ocupacao = Math.min(this.ocupacaoPorDia().get(k) ?? 0, 3)
      return {
        data: d, letra: diasLetras[d.getDay()],
        numero: d.getDate(), hoje: d.toDateString() === hoje.toDateString(),
        selecionado: d.toDateString() === this.diaSelecionado().toDateString(),
        ocupacao,
      }
    })
  }

  // Label do mês/ano da semana visível
  get mesLabel(): string {
    const dias = this.semana()
    const meio = dias[3].data  // quarta-feira da semana
    return meio.toLocaleDateString('pt-BR', { month: 'long', year: 'numeric' })
  }

    this.diaSelecionado.set(data)
    // Ajustar semanaOffset para a semana do dia selecionado
    const hoje = new Date()
    const diff  = Math.floor(

  }

ARQUIVO: agendamentos.repository.ts — adicionar métodos:

  async criar(payload: {
    profissional_id: string; servico_id: string; cliente_nome: string
    cliente_wpp: string; data_hora: string; status: string; observacoes?: string
  }): Promise<{ id: string }> {
    const { data, error } = await this.supabase
      .from('agendamentos').insert(payload).select('id').single()
    if (error) throw error
    return data
  }

  async atualizar(id: string, payload: Partial<{
    servico_id: string; cliente_nome: string; cliente_wpp: string
    data_hora: string; status: string; observacoes: string
  }>): Promise<void> {
    const profId = await this.profissionalIdOrThrow()
    const { error } = await this.supabase
      .from('agendamentos').update(payload)
      .eq('id', id).eq('profissional_id', profId)
    if (error) throw error
  }

  async excluir(id: string): Promise<void> {
    const profId = await this.profissionalIdOrThrow()
    const { error } = await this.supabase
      .from('agendamentos').delete()
      .eq('id', id).eq('profissional_id', profId)
    if (error) throw error
  }

    try {
      await this.servicosRepo.excluir(id)
      this.servicosStore.removerLocal(id)
      this.snackBar.open('Serviço removido.', '', { duration: 3000 })
    } catch (e: any) {
      // Postgres FK violation: código 23503
      if (e?.code === '23503' || e?.message?.includes('foreign key')) {
        this.snackBar.open(
          'Não é possível excluir este serviço pois ele possui agendamentos vinculados. ' +
          'Cancele ou conclua os agendamentos antes de excluir.',
          'Entendi',
          { duration: 8000, panelClass: 'snack-warn' }
        )
      } else {
        this.snackBar.open('Erro ao excluir serviço. Tente novamente.', '', { duration: 4000 })
        console.error('[Serviços] erro ao excluir:', e)
      }
    }
  }
