  import { Component, inject, signal, computed, OnInit } from '@angular/core'
  import { ActivatedRoute, Router } from '@angular/router'
  import { CommonModule } from '@angular/common'
  import { ReactiveFormsModule, FormBuilder, Validators } from '@angular/forms'
  import { MatSnackBar } from '@angular/material/snack-bar'
  import { AgendamentosStore } from '../../state/agendamentos.store'
  import { ServicosStore } from '../../state/servicos.store'
  import { AgendamentosRepository } from '../../../../core/repositories/agendamentos.repository'

  @Component({
    selector: 'app-agendamento-detalhe',
    standalone: true,
    imports: [CommonModule, ReactiveFormsModule, /* Material modules */],
    templateUrl: './agendamento-detalhe.component.html',
    styleUrl:    './agendamento-detalhe.component.scss',
  })
  export class AgendamentoDetalheComponent implements OnInit {
    private route   = inject(ActivatedRoute)
    private router  = inject(Router)
    private fb      = inject(FormBuilder)
    private snack   = inject(MatSnackBar)
    private agStore = inject(AgendamentosStore)
    private svStore = inject(ServicosStore)
    private agRepo  = inject(AgendamentosRepository)

    loading  = signal(false)
    salvando = signal(false)
    mostrarCampoMotivo = signal(false)

    agendamento = computed(() => {
      const id = this.route.snapshot.paramMap.get('id')
      return this.agStore.agendamentos().find(a => a.id === id) ?? null
    })

    servicos = computed(() => this.svStore.servicos().filter(s => s.ativo))

    form = this.fb.group({
      servico_id:   ['', Validators.required],
      data_hora:    ['', Validators.required],
      cliente_nome: ['', Validators.required],
      cliente_wpp:  ['', Validators.required],
      observacoes:  [''],
    })

    motivoRecusa = signal('')

    async ngOnInit(): Promise<void> {
      this.loading.set(true)
      if (!this.agStore.agendamentos().length) await this.agStore.carregar()
      if (!this.svStore.servicos().length)     await this.svStore.carregar()

      const ag = this.agendamento()
      if (!ag) { this.router.navigate(['/dashboard/agenda']); return }

      this.form.patchValue({
        servico_id:   ag.servico_id,
        data_hora:    ag.data_hora,
        cliente_nome: ag.cliente_nome,
        cliente_wpp:  ag.cliente_wpp,
        observacoes:  ag.observacoes ?? '',
      })
      this.loading.set(false)
    }

    voltar(): void {
      this.router.navigate(['/dashboard/agenda'])
    }

    async salvarEdicao(): Promise<void> {
      if (this.form.invalid) return
      this.salvando.set(true)
      try {
        await this.agRepo.atualizar(this.agendamento()!.id, this.form.value as any)
        this.agStore.carregar()
        this.snack.open('Agendamento atualizado.', '', { duration: 3000 })
        this.voltar()
      } catch (e) {
        this.snack.open('Erro ao salvar. Tente novamente.', '', { duration: 4000 })
      } finally {
        this.salvando.set(false)
      }
    }

    async confirmar(): Promise<void> {
      await this.executarAcao('confirmado')
    }

    async cancelar(): Promise<void> {
      await this.executarAcao('cancelado')
    }

    async confirmarRecusa(): Promise<void> {
      await this.executarAcao('recusado', this.motivoRecusa())
    }

    async naoCompareceu(): Promise<void> {
      await this.executarAcao('nao_compareceu')
    }

    reagendar(): void {
      const ag = this.agendamento()!
      // Buscar slug do profissional via currentUser signal
      const slug = this.agStore.profissionalSlug?.() ?? ''
      const link = `https://www.kianna.com.br/${slug}?reagendar=${ag.id}`
      navigator.clipboard.writeText(link)
      this.snack.open('Link de reagendamento copiado!', 'OK', { duration: 4000 })
    }

    private async executarAcao(
      status: 'confirmado' | 'recusado' | 'cancelado' | 'nao_compareceu',
      motivo?: string
    ): Promise<void> {
      this.salvando.set(true)
      try {
        await this.agRepo.atualizarStatus(this.agendamento()!.id, status, motivo)
        this.agStore.atualizarStatusLocal(this.agendamento()!.id, status)
        this.snack.open('Status atualizado.', '', { duration: 3000 })
        this.voltar()
      } catch (e) {
        this.snack.open('Erro ao atualizar. Tente novamente.', '', { duration: 4000 })
      } finally {
        this.salvando.set(false)
      }
    }
  }

  // Gerar semana com flag temSlots por dia
  private gerarSemanaComSlots(offset: number): DiaSemana[] {
    const hoje = new Date()
    const diasLetras = ['Dom','Seg','Ter','Qua','Qui','Sex','Sáb']
    const meses = ['jan','fev','mar','abr','mai','jun','jul','ago','set','out','nov','dez']
    const base = new Date(hoje)
    base.setDate(hoje.getDate() - hoje.getDay() + offset * 7)

    return Array.from({ length: 7 }, (_, i) => {
      const d = new Date(base)
      d.setDate(base.getDate() + i)
      const dataISO = d.toISOString().split('T')[0]
      const slots = this.calcularSlotsParaDia(dataISO)
      return {
        dataISO,
        letra:   diasLetras[d.getDay()],
        numero:  d.getDate(),
        mes:     meses[d.getMonth()],
        temSlots: slots.length > 0,
      }
    })
  }

--- E6: BUG CRÍTICO — "Sem horários disponíveis" mesmo com slots existentes ---

⚠️  ATENÇÃO CLAUDE CODE: INVESTIGAR ANTES DE IMPLEMENTAR

O cliente não consegue agendar mesmo com datas e horários configurados.
A mensagem "Sem horários disponíveis" aparece indevidamente.

Pontos a investigar no slot-calculator.service.ts:

  1. TIMEZONE
     Verificar se as datas estão sendo comparadas no timezone correto.
     Comparações entre Date objects UTC e strings locais podem causar
     o dia "errar" por 1 dia (ex: 2026-05-08T03:00:00Z vira 07/05 no UTC).
     Garantir que a comparação de dia da semana usa o timezone do profissional,
     não UTC.

  2. DIA DA SEMANA
     A tabela disponibilidades usa dia_semana (0=domingo...6=sábado).
     Verificar se new Date(dataISO).getDay() retorna o dia correto
     ou se está sendo afetado pelo timezone do browser vs UTC.
     Correção sugerida:
       // Em vez de:
       new Date(dataISO).getDay()
       // Usar:
       new Date(dataISO + 'T12:00:00').getDay()
       // (meio-dia evita virada de dia por timezone)

  3. INTERVALO DE BUSCA DE AGENDAMENTOS
     Verificar se a query de agendamentos confirmados está buscando
     o intervalo correto (hoje até hoje+30 dias).
     Se a query não retornar agendamentos, a lógica de bloqueio
     pode estar invertida (bloqueando tudo em vez de nada).

  4. SERVIÇO SEM DURAÇÃO
     Se duracao_min for null ou 0, o cálculo de slots pode gerar
     loop infinito ou array vazio. Adicionar guard:
       if (!servico?.duracao_min || servico.duracao_min <= 0) return []

  5. ANTECEDÊNCIA MÍNIMA
     Se antecedencia_minima_horas não existir ainda no banco
     (campo novo da task TASK_FLUXO_AGENDAMENTO_V2.txt),
     o valor pode ser undefined, causando comportamento inesperado.
     Adicionar fallback:
       const antecedencia = profissional.antecedencia_minima_horas ?? 24

  6. LOG DE DIAGNÓSTICO
     Adicionar logs temporários no slot-calculator para entender
     o que está sendo descartado:
       console.log('[SlotCalc] dia:', dataISO, '| dia semana:', diaSemana)
       console.log('[SlotCalc] disponibilidades encontradas:', disponibilidades.length)
       console.log('[SlotCalc] slots gerados antes de filtrar:', slots.length)
       console.log('[SlotCalc] slots após filtro de antecedência:', slotsValidos.length)
       console.log('[SlotCalc] slots após filtro de capacidade:', slotsFinal.length)

  Após corrigir o bug, remover os logs de diagnóstico.
