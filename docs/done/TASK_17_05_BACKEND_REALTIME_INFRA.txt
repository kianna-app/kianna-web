--- A3: Módulo de Auth (validação do JWT Supabase) ---

ARQUIVO: src/modules/auth/auth.guard.ts

  O backend recebe o JWT do Supabase no header Authorization.
  Validar com supabase.auth.getUser(token):

  @Injectable()
  export class SupabaseAuthGuard implements CanActivate {
    constructor(private config: ConfigService) {}

    async canActivate(context: ExecutionContext): Promise<boolean> {
      const req = context.switchToHttp().getRequest()
      const token = req.headers.authorization?.replace('Bearer ', '')
      if (!token) throw new UnauthorizedException()

      const supabase = createClient(
        this.config.get('SUPABASE_URL'),
        this.config.get('SUPABASE_ANON_KEY')
      )
      const { data, error } = await supabase.auth.getUser(token)
      if (error || !data.user) throw new UnauthorizedException()

      req.user = data.user
      return true
    }
  }

  @Injectable()
  export class RolesGuard implements CanActivate {
    constructor(private reflector: Reflector, private config: ConfigService) {}

    async canActivate(context: ExecutionContext): Promise<boolean> {
      const roles = this.reflector.get<string[]>('roles', context.getHandler())
      if (!roles) return true // sem decorator = acesso livre

      const req = context.switchToHttp().getRequest()
      const userId = req.user?.id
      if (!userId) return false

      const supabase = createSupabaseClient(this.config)
      const { data } = await supabase
        .from('profissionais')
        .select('role')
        .eq('user_id', userId)
        .single()

      return roles.includes(data?.role)
    }
  }

--- B2: Subscription Realtime ---

ARQUIVO: header.component.ts (continuação)

  private channel: any

  private subscribeRealtime(): void {
    const profId = this.currentUser()?.profissional_id
    if (!profId) return

    this.channel = this.supabase
      .channel('pendentes-realtime')
      .on(
        'postgres_changes',
        {
          event: '*',     // INSERT, UPDATE, DELETE
          schema: 'public',
          table: 'agendamentos',
          filter: `profissional_id=eq.${profId}`
        },
        (_payload: any) => {
          // Qualquer mudança → recontar
          this.carregarPendentes()
        }
      )
      .subscribe()
  }

  ngOnDestroy(): void {
    if (this.channel) {
      this.supabase.removeChannel(this.channel)
    }
  }

  import { AbstractControl, ValidationErrors, ValidatorFn, Validators } from '@angular/forms'

  export class KiannaValidators {

    /** Nome: mín 2 chars, sem números */
    static nome(): ValidatorFn[] {
      return [
        Validators.required,
        Validators.minLength(2),
        Validators.maxLength(100),
        (c: AbstractControl): ValidationErrors | null =>
          c.value && /\d/.test(c.value) ? { nomeInvalido: true } : null
      ]
    }

    /** WhatsApp: 10-11 dígitos, DDD obrigatório */
    static whatsapp(): ValidatorFn[] {
      return [
        Validators.required,
        (c: AbstractControl): ValidationErrors | null => {
          const digits = (c.value || '').replace(/\D/g, '')
          if (digits.length < 10 || digits.length > 11) return { wppInvalido: true }
          return null
        }
      ]
    }

    /** Preço: >= 0, máx 2 decimais */
    static preco(): ValidatorFn[] {
      return [
        Validators.required,
        Validators.min(0),
        (c: AbstractControl): ValidationErrors | null =>
          c.value && !/^\d+(\.\d{1,2})?$/.test(String(c.value))
            ? { precoInvalido: true } : null
      ]
    }

    /** Duração em minutos: 5-480 */
    static duracao(): ValidatorFn[] {
      return [Validators.required, Validators.min(5), Validators.max(480)]
    }

    /** Email */
    static email(): ValidatorFn[] {
      return [Validators.required, Validators.email]
    }

    /** Slug: lowercase, hifens, sem espaços */
    static slug(): ValidatorFn[] {
      return [
        Validators.required,
        Validators.minLength(3),
        Validators.maxLength(60),
        Validators.pattern(/^[a-z0-9]+(-[a-z0-9]+)*$/)
      ]
    }
  }

  export const ERROR_MESSAGES: Record<string, string | ((params: any) => string)> = {
    required:      'Campo obrigatório.',
    minlength:     (p) => `Mínimo ${p.requiredLength} caracteres.`,
    maxlength:     (p) => `Máximo ${p.requiredLength} caracteres.`,
    min:           (p) => `Valor mínimo: ${p.min}.`,
    max:           (p) => `Valor máximo: ${p.max}.`,
    email:         'E-mail inválido.',
    pattern:       'Formato inválido.',
    nomeInvalido:  'Nome não pode conter números.',
    wppInvalido:   'WhatsApp inválido. Informe DDD + número (10 ou 11 dígitos).',
    precoInvalido: 'Preço inválido. Use até 2 casas decimais.',
  }

  export function getErrorMessage(control: AbstractControl): string {
    if (!control.errors) return ''
    const key = Object.keys(control.errors)[0]
    const msg = ERROR_MESSAGES[key]
    if (!msg) return 'Campo inválido.'
    return typeof msg === 'function' ? msg(control.errors[key]) : msg
  }

--- E1: Ativar Point-in-Time Recovery (PITR) ---

O Supabase free não tem backup automático.
Para backups automáticos precisa do plano Pro ($25/mês).

OPÇÃO A — Plano Pro Supabase (recomendado para produção):
  1. Painel Supabase → Organization → Billing → Upgrade to Pro
  2. Após upgrade: Database → Backups → PITR habilitado automaticamente
  3. Retenção: 7 dias de backups contínuos
  4. Restore: qualquer ponto nos últimos 7 dias

OPÇÃO B — Backup manual (plano free, pré-lançamento):
  1. Painel Supabase → Database → Backups
  2. Clicar em "Download backup" periodicamente (semanal)
  3. Salvar em local seguro (Google Drive, S3)
