ARQUIVO: src/modules/booking/booking.controller.ts (novo módulo)

@ApiTags('Booking')
@Controller('api/booking')
export class BookingController {
  constructor(
    private profService: ProfissionaisService,
    private servicosService: ServicosService,
    private dispService: DisponibilidadesService,
    private bloqueiosService: BloqueiosService,
  ) {}

  @Get(':slug')
  @ApiOperation({ summary: 'Dados completos para página de booking' })
  async dadosBooking(@Param('slug') slug: string) {
    const prof = await this.profService.buscarPorSlug(slug)
    if (!prof) throw new NotFoundException('Profissional não encontrado')

    const hoje = new Date().toISOString().split('T')[0]
    const janela = prof.antecedencia_maxima_dias ?? 30
    const dataFim = new Date()
    dataFim.setDate(dataFim.getDate() + janela)
    const dataFimISO = dataFim.toISOString().split('T')[0]

    const [servicos, disponibilidades, bloqueios] = await Promise.all([
      this.servicosService.listarAtivos(prof.id),
      this.dispService.listarPorProfissional(prof.id),
      this.bloqueiosService.listarPorPeriodo(prof.id, hoje, dataFimISO),
    ])

    return {
      profissional: {
        id: prof.id,
        nome: prof.nome,
        slug: prof.slug,
        foto_url: prof.foto_url,
        especialidade: prof.especialidade,
        bio: prof.bio,
        whatsapp: prof.whatsapp,
        antecedencia_minima_horas: prof.antecedencia_minima_horas,
        antecedencia_maxima_dias: prof.antecedencia_maxima_dias,
        timezone: prof.timezone,
      },
      servicos,
      disponibilidades,
      bloqueios,
    }
  }
}
