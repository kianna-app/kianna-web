LÓGICA — visao-geral.component.ts:
  // Adicionar computed signals:
  readonly pendentes = computed(() =>
    this.agendamentosStore.agendamentos().filter(a => a.status === 'pendente')
  )
  readonly agendamentosHoje = computed(() => {
    const hoje = new Date().toDateString()
    return this.agendamentosStore.agendamentos().filter(a =>
      new Date(a.data_hora).toDateString() === hoje &&
      a.status !== 'cancelado'
    )
  })

  // Método confirmar (atualiza store e muda status localmente):
  async confirmar(id: string): Promise<void> {
    // marcar loading no item
    try {
      await this.agendamentosRepo.atualizarStatus(id, 'confirmado')
      // atualizar item no store localmente (sem recarregar tudo):
      this.agendamentosStore.atualizarStatusLocal(id, 'confirmado')
    } catch (e) {
      console.error('[VisaoGeral] erro ao confirmar:', e)
    }
  }

  async cancelar(id: string): Promise<void> {
    try {
      await this.agendamentosRepo.atualizarStatus(id, 'cancelado')
      this.agendamentosStore.atualizarStatusLocal(id, 'cancelado')
    } catch (e) {
      console.error('[VisaoGeral] erro ao cancelar:', e)
    }
  }

  // agendamentos.store.ts — adicionar método:
  atualizarStatusLocal(id: string, status: string): void {
    this.agendamentos.update(lista =>
      lista.map(a => a.id === id ? { ...a, status } : a)
    )
  }

  // agendamentos.repository.ts — garantir método (ver fix anterior A3):
  async atualizarStatus(id: string, status: 'confirmado' | 'cancelado'): Promise<void> {
    const profId = await this.profissionalIdOrThrow()
    const { error } = await this.supabase
      .from('agendamentos')
      .update({ status })
      .eq('id', id)
      .eq('profissional_id', profId)
    if (error) throw error
  }
