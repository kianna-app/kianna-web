--- A4: INSERT agendamento retorna 401 (RLS bloqueando anon) ---
CAUSA CONFIRMADA: policy de INSERT para anon não existe ou está errada.
Erro no log: "new row violates row-level security policy for table agendamentos"

AÇÃO — Rodar no SQL Editor do Supabase:

-- Remover policy conflitante se existir
DROP POLICY IF EXISTS "anon pode criar agendamento" ON agendamentos;

-- Criar policy correta
CREATE POLICY "anon pode criar agendamento"
ON agendamentos
FOR INSERT
TO anon
WITH CHECK (
  profissional_id IN (
    SELECT id FROM profissionais WHERE ativo = true
  )
);

-- Garantir que RLS está ativo na tabela
ALTER TABLE agendamentos ENABLE ROW LEVEL SECURITY;

-- Verificar policies existentes
SELECT policyname, cmd, roles, qual, with_check
FROM pg_policies
WHERE tablename = 'agendamentos';

--- A5: View agendamentos_publicos com SECURITY DEFINER ---
CAUSA: view criada com SECURITY DEFINER executa com permissões do criador, não do usuário.
AÇÃO — Recriar a view sem SECURITY DEFINER:

DROP VIEW IF EXISTS agendamentos_publicos;

CREATE VIEW agendamentos_publicos
WITH (security_invoker = true)
AS
  SELECT id, profissional_id, servico_id, data_hora, status
  FROM agendamentos
  WHERE status IN ('pendente', 'confirmado');

-- Policy de SELECT na view para anon
CREATE POLICY "anon pode ver agendamentos publicos"
ON agendamentos
FOR SELECT
TO anon
USING (status IN ('pendente', 'confirmado'));
