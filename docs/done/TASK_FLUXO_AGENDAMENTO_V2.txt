--- A4: Tabela bloqueios ---

-- Bloqueios de agenda: dia inteiro ou período específico numa data específica
CREATE TABLE IF NOT EXISTS bloqueios (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  profissional_id UUID NOT NULL REFERENCES profissionais(id) ON DELETE CASCADE,
  data          DATE NOT NULL,                          -- data do bloqueio
  hora_inicio   TIME,                                   -- NULL = dia inteiro
  hora_fim      TIME,                                   -- NULL = dia inteiro
  motivo        TEXT,                                   -- opcional
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),

  -- Se hora_inicio for informado, hora_fim também deve ser
  CONSTRAINT bloqueios_horas_check CHECK (
    (hora_inicio IS NULL AND hora_fim IS NULL) OR
    (hora_inicio IS NOT NULL AND hora_fim IS NOT NULL AND hora_fim > hora_inicio)
  )
);

-- Índice para busca por profissional e data
CREATE INDEX IF NOT EXISTS idx_bloqueios_profissional_data
  ON bloqueios(profissional_id, data);

-- RLS
ALTER TABLE bloqueios ENABLE ROW LEVEL SECURITY;

-- Profissional autenticado gerencia seus próprios bloqueios
CREATE POLICY "profissional gerencia bloqueios"
ON bloqueios FOR ALL TO authenticated
USING (
  profissional_id IN (SELECT id FROM profissionais WHERE user_id = auth.uid())
)
WITH CHECK (
  profissional_id IN (SELECT id FROM profissionais WHERE user_id = auth.uid())
);

-- Anon pode ver bloqueios (necessário para slot-calculator da página pública)
CREATE POLICY "anon pode ver bloqueios"
ON bloqueios FOR SELECT TO anon
USING (true);

  async ngOnInit(): Promise<void> {
    // ... inicialização existente ...

    // Finalizar agendamentos confirmados vencidos
    await this.finalizarVencidos()
  }

  private async finalizarVencidos(): Promise<void> {
    const profId = this.currentUser()?.profissional_id
    if (!profId) return

    const agora = new Date()
    const vencidos = this.agendamentosStore.agendamentos().filter(a => {
      if (a.status !== 'confirmado') return false
      const servico = this.servicosStore.servicos().find(s => s.id === a.servico_id)
      const duracaoMin = servico?.duracao_min ?? 0
      const fimAgendamento = new Date(new Date(a.data_hora).getTime() + duracaoMin * 60000)
      return fimAgendamento < agora
    })

    for (const ag of vencidos) {
      await this.agendamentosRepo.atualizarStatus(ag.id, 'finalizado')
      this.agendamentosStore.atualizarStatusLocal(ag.id, 'finalizado')
    }
  }

  // NOTA: executa silenciosamente, sem confirmação do profissional.
  // Executar após carregar agendamentos e serviços.
